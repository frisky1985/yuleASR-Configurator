复跑命令（一次跑完 dry-run → apply → build → ctest → rollback）：

```bash
# 1) 建 scratch 副本（已跟踪文件 + git init + 基线 commit）
REAL=~/.openclaw/workspace/yuleASR; SCRATCH=/tmp/ver002-scratch; OUT=/tmp/ver002-out; BUILD=/tmp/ver002-build
rm -rf $SCRATCH $OUT $BUILD; mkdir -p $SCRATCH $OUT
git -C $REAL ls-files | tar -T - -C $REAL -cf - | tar -C $SCRATCH -xf -

YULEASR_DIR=$SCRATCH REPLACE_OUT=$OUT REPLACE_MODE=dry-run \
  npx vitest run apps/yuleasr-web/src/services/__tests__/replace-cfgh-run.test.ts
YULEASR_DIR=$SCRATCH REPLACE_OUT=$OUT REPLACE_MODE=apply \
  npx vitest run apps/yuleasr-web/src/services/__tests__/replace-cfgh-run.test.ts

# 3) 编译 + 测试
cmake -B $BUILD -S $SCRATCH -DCMAKE_BUILD_TYPE=Debug -DBUILD_TESTING=ON

(cd $BUILD && ctest --output-on-failure)

# 4) rollback + 归零校验
YULEASR_DIR=$SCRATCH REPLACE_OUT=$OUT REPLACE_MODE=rollback \
  npx vitest run apps/yuleasr-web/src/services/__tests__/replace-cfgh-run.test.ts
git -C $SCRATCH status --porcelain | wc -l   # 必须为 0
git -C $REAL status --porcelain | wc -l      # 必须为 0（真实仓库只读）
```
